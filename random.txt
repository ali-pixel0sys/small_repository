import random

def word_guessing_game():
    """A simple interactive word guessing game in Python."""
    words = ["python", "coder", "logic", "matrix", "binary", "syntax"]
    secret_word = random.choice(words)
    attempts = 5
    
    print("=== Mini Word Guessing Game ===")
    print(f"Guess the 6-letter secret word! You have {attempts} attempts.\n")
    
    for attempt in range(1, attempts + 1):
        guess = input(f"Attempt {attempt}/{attempts} - Enter your guess: ").strip().lower()
        
        if len(guess) != len(secret_word):
            print(f"⚠️ Word must be exactly {len(secret_word)} letters long!\n")
            continue
            
        if guess == secret_word:
            print(f"\n🎉 Congratulations! You guessed the word '{secret_word}' correctly!")
            return
            
        # Provide matching letter feedback
        feedback = "".join([c if c == secret_word[i] else "_" for i, c in enumerate(guess)])
        print(f"Feedback: {feedback}\n")
        
    print(f"❌ Game Over! The secret word was '{secret_word}'.")

if __name__ == "__main__":
    word_guessing_game()
